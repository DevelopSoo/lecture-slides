STEP 1

파일 만들고 함수 내보내기

import { NextResponse, NextRequest } from "next/server";

export function proxy(request: NextRequest) {
  return NextResponse.next();
}

next() = 아무것도 막지 않고 그대로 통과

STEP 2

로그인 안 했으면 돌려보내기

실습이라 false로 고정해둔 값입니다. 실제로는 쿠키에서 로그인 여부를 꺼내 확인합니다.

import { NextResponse, NextRequest } from "next/server";

export function proxy(request: NextRequest) {
  const isAuthenticated = false;

  if (!isAuthenticated) {
    return NextResponse.redirect(new URL("/login", request.url));
  }

  return NextResponse.next();
}

STEP 3

어느 주소에서 실행할지

export const config = {
  matcher: "/profile/:path*",
};

이 줄이 없으면 모든 페이지에서 실행됩니다

완성 코드

// src/proxy.ts
import { NextResponse, NextRequest } from "next/server";

export function proxy(request: NextRequest) {
  const isAuthenticated = false;

  if (!isAuthenticated) {
    return NextResponse.redirect(new URL("/login", request.url));
  }

  return NextResponse.next();
}

export const config = {
  matcher: "/profile/:path*",
};
